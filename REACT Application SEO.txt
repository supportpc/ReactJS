import type { Metadata } from "next";

export async function generateMetadata({
  params,
}: {
  params: Promise<{ slug: string }>;
}): Promise<Metadata> {
  const { slug } = await params;

  const page = await getPage(slug);

  return {
    title: page.metaTitle,
    description: page.metaDescription,
    keywords: page.metaKeywords,
    openGraph: {
      title: page.metaTitle,
      description: page.metaDescription,
      images: [page.img],
    },
  };
}

export default async function Page({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;

  const page = await getPage(slug);

  return (
    <main>
      <h1>{page.h1}</h1>
    </main>
  );
}

•	SSG
•	SSR
•	backend

5. Gatsby

Gatsby is also React-based, but its philosophy historically focused heavily on static/content sites. Gatsby now supports SSG, DSG and SSR on a per-page basis
Good for:
•	Documentation
•	Blogs
•	Content websites
•	Marketing sites
•	Static websites
•	Sites consuming CMS/API data
•	CMS
•	 ↓
•	Gatsby
•	 ↓
•	Build
•	 ↓
•	HTML/CSS/JS
•	 ↓
•	CDN

Gatsby's SSG output can be served as static assets, while DSG can defer generation and SSR can render at request time. 

For a brand-new general-purpose React application, however, I'd usually choose Next.js or React + Vite before Gatsby.

10. Angular
Angular is different from React.
Angular is much more of a complete frontend framework.
It includes a lot of things out of the box:
•	Components
•	Routing
•	Dependency injection
•	Forms
•	HTTP client
•	Testing patterns
•	State/service architecture
•	TypeScript-first development
It's particularly common in:
•	Enterprise applications
•	Large business systems
•	Corporate applications
•	Complex internal applications
For example:
Banking dashboard

11. Ember.js

Ember
 ↓
