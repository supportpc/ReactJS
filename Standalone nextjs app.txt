# 5. Add Linux sharp (copy the lines below exactly; -ErrorAction SilentlyContinue stays on ONE line)

$proj = (Get-Location).Path
$v = (Get-Content node_modules\sharp\package.json | ConvertFrom-Json).version
Remove-Item $env:TEMP\sharp-linux -Recurse -Force -ErrorAction SilentlyContinue
mkdir $env:TEMP\sharp-linux
cd $env:TEMP\sharp-linux
npm init -y
npm install --os=linux --cpu=x64 --libc=glibc sharp@$v
xcopy node_modules\@img "$proj\.next\standalone\node_modules\@img" /E /I /Y
cd $proj
