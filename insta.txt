            // ---------- INSTAGRAM FEED (with demo data fallback if token fails) ----------
            async function fetchInstagramFeed(token) {
                const apiUrl = `https://graph.facebook.com/v20.0/17841461859295137/media?fields=id,media_type,media_url,permalink,like_count,timestamp&access_token=${token}`;
                try {
                    let response = await fetch(apiUrl);
                    let data = await response.json();

                    if (data.data) {
                        data.data = data.data.filter(
                            post => typeof post.media_url === "string" && post.media_url.trim() !== ""
                        );
                    }

                    if (data.error && data.error.code === 190) {  // token expired
                        console.warn("Token expired, use demo feed");
                        return null;  // will trigger demo
                    }
                    return data;
                } catch (e) {
                    console.warn("Fetch error, using demo feed");
                    return null;
                }
            }

            // ---------- DRAG TO SCROLL (mouse drag) ----------
            function initDragScroll() {
                const gallery = document.querySelector('.gallery');
                if (!gallery) return;

                let isDown = false;
                let startX;
                let scrollLeft;

                let current = 0;
                let target = 0;
                const ease = 0.1;

                gallery.addEventListener('mousedown', (e) => {
                    isDown = true;
                    gallery.classList.add('active'); // for grabbing
                    startX = e.pageX - gallery.offsetLeft;
                    scrollLeft = gallery.scrollLeft;
                    // prevent default to avoid text selection while dragging
                    e.preventDefault();
                });

                gallery.addEventListener('mouseleave', () => {
                    isDown = false;
                    gallery.classList.remove('active');
                });

                gallery.addEventListener('mouseup', () => {
                    isDown = false;
                    gallery.classList.remove('active');
                });

                gallery.addEventListener('mousemove', (e) => {
                    if (!isDown) return;
                    e.preventDefault();
                    const x = e.pageX - gallery.offsetLeft;
                    const walk = (x - startX) * 2.1; // scroll speed multiplier
                    gallery.scrollLeft = scrollLeft - walk;
                });

                // optional: disable drag on clicks on links (they will still work after drag end)
                gallery.querySelectorAll('a').forEach(link => {
                    link.addEventListener('dragstart', (e) => e.preventDefault());
                });
            }

            // ---------- ARROW BUTTONS ----------
            function initArrowButtons() {
                const leftBtn = document.getElementById('arrow-left');
                const rightBtn = document.getElementById('arrow-right');
                const gallery = document.querySelector('.gallery');

                if (!leftBtn || !rightBtn || !gallery) return;

                leftBtn.addEventListener('click', () => {
                    gallery.scrollBy({left: -300, behavior: 'smooth'});
                });

                rightBtn.addEventListener('click', () => {
                    gallery.scrollBy({left: 300, behavior: 'smooth'});
                });
            }

            // start everything after DOM ready
            document.addEventListener('DOMContentLoaded', async () => {
                await loadInstagramFeed();
                initDragScroll();   // 👈 mouse drag enabled
                initArrowButtons(); // 👈 left/right arrows
            });
